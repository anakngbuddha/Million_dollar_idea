/**
 * Fixed Cloudinary Download Utilities
 * For unsigned uploads with cloud name: duqkmbepu
 */

const CLOUD_NAME = 'duqkmbepu';

/**
 * Extract public ID from Cloudinary URL
 */
export const extractPublicIdFromUrl = (url: string): string | null => {
  try {
    // Pattern: https://res.cloudinary.com/{cloud}/resource_type/upload/v{version}/{folder}/{publicId}.{ext}
    const uploadIndex = url.indexOf('/upload/');
    if (uploadIndex === -1) return null;
    
    const afterUpload = url.substring(uploadIndex + 8); // Skip '/upload/'
    
    // Remove version number if present (v1234567890/)
    const withoutVersion = afterUpload.replace(/^v\d+\//, '');
    
    // Remove file extension and query params
    const publicId = withoutVersion.split('?')[0].replace(/\.[^.]+$/, '');
    
    return publicId;
  } catch (error) {
    console.error('Error extracting public ID:', error);
    return null;
  }
};

/**
 * Get resource type from Cloudinary URL
 */
export const getResourceType = (url: string): string => {
  if (url.includes('/image/upload')) return 'image';
  if (url.includes('/video/upload')) return 'video';
  if (url.includes('/raw/upload')) return 'raw';
  return 'raw'; // default for documents
};

/**
 * Format file size
 */
export const formatFileSize = (bytes: number): string => {
  if (bytes === 0) return '0 Bytes';
  const k = 1024;
  const sizes = ['Bytes', 'KB', 'MB', 'GB'];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return Math.round((bytes / Math.pow(k, i)) * 100) / 100 + ' ' + sizes[i];
};

/**
 * Generate Cloudinary download URL with forced download
 * This is the KEY function for making downloads work properly
 */
export const getCloudinaryDownloadUrl = (
  publicId: string,
  fileName: string,
  resourceType: string = 'raw',
  format?: string
): string => {
  // Clean public ID
  const cleanPublicId = publicId.trim().replace(/^\/+|\/+$/g, '');
  
  // Encode filename for URL safety
  const safeFileName = encodeURIComponent(fileName);
  
  // Build URL with fl_attachment to force download
  let url = `https://res.cloudinary.com/${CLOUD_NAME}/${resourceType}/upload/fl_attachment:${safeFileName}/${cleanPublicId}`;
  
  // Add format extension if provided
  if (format) {
    url += `.${format}`;
  }
  
  console.log('Generated download URL:', url);
  return url;
};

/**
 * Convert existing Cloudinary URL to download URL
 */
export const convertSecureUrlToDownloadUrl = (secureUrl: string, fileName?: string): string | null => {
  try {
    console.log('Converting URL:', secureUrl);
    
    // Extract public ID and resource type
    const publicId = extractPublicIdFromUrl(secureUrl);
    const resourceType = getResourceType(secureUrl);
    
    if (!publicId) {
      console.error('Could not extract public ID from URL');
      return null;
    }
    
    console.log('Extracted public ID:', publicId);
    console.log('Resource type:', resourceType);
    
    // Extract format from original URL
    const formatMatch = secureUrl.match(/\.([a-zA-Z0-9]+)(?:\?|$)/);
    const format = formatMatch ? formatMatch[1] : undefined;
    
    // Get filename from URL or use provided filename
    const urlFileName = fileName || secureUrl.split('/').pop()?.split('?')[0] || 'download';
    
    // Generate download URL
    return getCloudinaryDownloadUrl(publicId, urlFileName, resourceType, format);
  } catch (error) {
    console.error('Error converting URL:', error);
    return null;
  }
};

/**
 * Download file using publicId (RECOMMENDED METHOD)
 */
export const downloadFileByPublicId = (
  publicId: string,
  fileName: string,
  resourceType: string = 'raw',
  format?: string
): void => {
  try {
    console.log('=== Download by Public ID ===');
    console.log('Public ID:', publicId);
    console.log('File Name:', fileName);
    console.log('Resource Type:', resourceType);
    console.log('Format:', format);
    
    const downloadUrl = getCloudinaryDownloadUrl(publicId, fileName, resourceType, format);
    console.log('Download URL:', downloadUrl);
    
    // Create temporary link and click it
    const link = document.createElement('a');
    link.href = downloadUrl;
    link.download = fileName;
    link.setAttribute('target', '_blank');
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    
    console.log('✅ Download initiated');
  } catch (error) {
    console.error('❌ Download error:', error);
    alert(`Failed to download file: ${error instanceof Error ? error.message : 'Unknown error'}`);
  }
};

/**
 * Download file using URL (FALLBACK METHOD)
 */
export const downloadFile = (fileUrl: string, fileName: string = 'download'): void => {
  try {
    console.log('=== Download by URL ===');
    console.log('URL:', fileUrl);
    console.log('File Name:', fileName);
    
    // Check if it's a Cloudinary URL
    if (fileUrl.includes('res.cloudinary.com')) {
      // Try to convert to download URL
      const downloadUrl = convertSecureUrlToDownloadUrl(fileUrl, fileName);
      
      if (downloadUrl) {
        console.log('Using converted download URL');
        const link = document.createElement('a');
        link.href = downloadUrl;
        link.download = fileName;
        link.setAttribute('target', '_blank');
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
        return;
      }
    }
    
    // Fallback: open URL directly
    console.log('Using direct URL');
    window.open(fileUrl, '_blank');
    
  } catch (error) {
    console.error('❌ Download error:', error);
    alert(`Failed to download file: ${error instanceof Error ? error.message : 'Unknown error'}`);
  }
};

/**
 * Get downloadable URL (helper function)
 */
export const getDownloadUrl = (
  fileUrl: string, 
  fileName: string = 'download', 
  publicId?: string,
  resourceType?: string,
  format?: string
): string => {
  if (publicId) {
    return getCloudinaryDownloadUrl(publicId, fileName, resourceType || 'raw', format);
  }
  return convertSecureUrlToDownloadUrl(fileUrl, fileName) || fileUrl;
};
