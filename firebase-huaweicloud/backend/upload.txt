/**
 * Fixed Cloudinary Upload Service
 * Cloud name: duqkmbepu
 * Using unsigned upload preset
 */

const CLOUD_NAME = 'duqkmbepu';
const UPLOAD_PRESET = 'your_unsigned_preset_name'; // REPLACE WITH YOUR ACTUAL PRESET NAME

export interface UploadResponse {
  publicId: string;
  fileName: string;
  fileUrl: string;
  fileSize: number;
  resourceType: string;
  format: string;
}

/**
 * Validate file before upload
 */
export const validateFile = (
  file: File,
  maxSizeMB: number = 100,
  allowedTypes: string[] = []
): { valid: boolean; error?: string } => {
  // Check file size
  const maxSizeBytes = maxSizeMB * 1024 * 1024;
  if (file.size > maxSizeBytes) {
    return {
      valid: false,
      error: `File size exceeds ${maxSizeMB}MB limit`
    };
  }

  // If no type restrictions, allow all
  if (allowedTypes.length === 0) {
    return { valid: true };
  }

  // Check file type
  const fileExtension = file.name.split('.').pop()?.toLowerCase();
  if (!fileExtension || !allowedTypes.includes(fileExtension)) {
    return {
      valid: false,
      error: `File type .${fileExtension} is not allowed`
    };
  }

  return { valid: true };
};

/**
 * Determine resource type based on file
 */
const getResourceTypeFromFile = (file: File): string => {
  const type = file.type.toLowerCase();
  
  if (type.startsWith('image/')) return 'image';
  if (type.startsWith('video/')) return 'video';
  
  // Everything else is 'raw' (documents, PDFs, etc.)
  return 'raw';
};

/**
 * Upload file to Cloudinary using unsigned upload
 */
export const uploadFile = async (
  file: File,
  classroomId: string,
  assignmentId: string,
  submissionId?: string
): Promise<UploadResponse> => {
  try {
    console.log('=== Starting Cloudinary Upload ===');
    console.log('File:', file.name);
    console.log('Size:', file.size);
    console.log('Type:', file.type);

    // Build folder path
    const folder = submissionId 
      ? `assignments/${classroomId}/${assignmentId}/submissions/${submissionId}`
      : `assignments/${classroomId}/${assignmentId}`;

    console.log('Upload folder:', folder);

    // Determine resource type
    const resourceType = getResourceTypeFromFile(file);
    console.log('Resource type:', resourceType);

    // Create form data
    const formData = new FormData();
    formData.append('file', file);
    formData.append('upload_preset', UPLOAD_PRESET);
    formData.append('folder', folder);
    formData.append('resource_type', resourceType);

    // Upload to Cloudinary
    const uploadUrl = `https://api.cloudinary.com/v1_1/${CLOUD_NAME}/${resourceType}/upload`;
    console.log('Upload URL:', uploadUrl);

    const response = await fetch(uploadUrl, {
      method: 'POST',
      body: formData
    });

    if (!response.ok) {
      const errorData = await response.json();
      console.error('Upload failed:', errorData);
      throw new Error(errorData.error?.message || 'Upload failed');
    }

    const data = await response.json();
    console.log('Upload response:', data);

    // Extract important data
    const uploadResponse: UploadResponse = {
      publicId: data.public_id,
      fileName: file.name,
      fileUrl: data.secure_url,
      fileSize: data.bytes,
      resourceType: data.resource_type,
      format: data.format
    };

    console.log('✅ Upload successful:', uploadResponse);
    return uploadResponse;

  } catch (error) {
    console.error('❌ Upload error:', error);
    throw new Error(error instanceof Error ? error.message : 'Upload failed');
  }
};

/**
 * Upload multiple files
 */
export const uploadMultipleFiles = async (
  files: File[],
  classroomId: string,
  assignmentId: string,
  submissionId?: string
): Promise<UploadResponse[]> => {
  const uploadPromises = files.map(file => 
    uploadFile(file, classroomId, assignmentId, submissionId)
  );
  
  return Promise.all(uploadPromises);
};
